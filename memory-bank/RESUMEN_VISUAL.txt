│                        🚀 ENDPOINTS DISPONIBLES (5)                          │
└──────────────────────────────────────────────────────────────────────────────┘

✅ GET /api/user
   └─ Listar usuarios con paginación
   └─ Parámetros: take=10, skip=0, search=...
   └─ Retorna: Lista paginada + info paginación
   └─ Status: 200

✅ POST /api/user
   └─ Crear nuevo usuario
   └─ Incluye: validación, asignación de rol, sincronización de permisos
   └─ Requiere: name, email, password, active, id_rol
   └─ Status: 201

✅ GET /api/user/{id}
   └─ Obtener usuario específico
   └─ Retorna: datos completos + roles + permisos
   └─ Status: 200 / 404

✅ PUT /api/user/{id}
   └─ Actualizar usuario
   └─ Actualización parcial (todos campos opcionales)
   └─ Sincroniza rol y permisos si se envía id_rol
   └─ Status: 200 / 404

✅ DELETE /api/user/{id}
   └─ Eliminar usuario
   └─ Previene auto-eliminación
   └─ Limpia roles, permisos y tokens
   └─ Status: 200 / 403 / 404

┌──────────────────────────────────────────────────────────────────────────────┐
│                     🧪 EJEMPLOS DE USO RÁPIDO                               │
└──────────────────────────────────────────────────────────────────────────────┘

1️⃣ LISTAR USUARIOS
   curl -X GET "http://paneladmin.local/api/user?take=10&skip=0" \
     -H "Authorization: Bearer TOKEN"

2️⃣ CREAR USUARIO
   curl -X POST "http://paneladmin.local/api/user" \
     -H "Authorization: Bearer TOKEN" \
     -H "Content-Type: application/json" \
     -d '{
       "name": "John Doe",
       "email": "john@example.com",
       "password": "SecurePass123",
       "password_confirmation": "SecurePass123",
       "active": 1,
       "id_rol": 2
     }'

3️⃣ OBTENER USUARIO
   curl -X GET "http://paneladmin.local/api/user/1" \
     -H "Authorization: Bearer TOKEN"

4️⃣ ACTUALIZAR USUARIO
   curl -X PUT "http://paneladmin.local/api/user/1" \
     -H "Authorization: Bearer TOKEN" \
     -H "Content-Type: application/json" \
     -d '{"name": "Jane Doe", "id_rol": 3}'

5️⃣ ELIMINAR USUARIO
   curl -X DELETE "http://paneladmin.local/api/user/10" \
     -H "Authorization: Bearer TOKEN"
